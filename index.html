<!-- ASMR Engine for Elektromir (Soft Pastel Blue Edition) -->
<script>
class ElektromirASMR {
  static start(options = {}) {
    if (window._elektromirASMR) return window._elektromirASMR;
    window._elektromirASMR = new ElektromirASMR(options);
    return window._elektromirASMR;
  }

  constructor(opts) {
    this.opt = Object.assign({ season: 'autumn', volume: 0.45, clicks: true, scroll: true }, opts);
    this.ctx = null;
    this.master = null;
    this.bindEvents();
  }

  initAudio() {
    if (!this.ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      this.ctx = new AC();
      this.master = this.ctx.createGain();
      this.master.gain.setValueAtTime(this.opt.volume, this.ctx.currentTime);
      this.master.connect(this.ctx.destination);
    }
    if (this.ctx.state === 'suspended') this.ctx.resume();
  }

  // Приятные бархатные щелчки при касаниях
  playClick(type = 'wood') {
    this.initAudio();
    const t = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    if (type === 'wood') {
      // Теплый клик дерева (Marimba)
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(520, t);
      osc.frequency.exponentialRampToValueAtTime(110, t + 0.035);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.28, t + 0.002);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.04);
      osc.connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.045);
    } else if (type === 'drop') {
      // Чистая капля росы
      osc.type = 'sine';
      osc.frequency.setValueAtTime(360, t);
      osc.frequency.exponentialRampToValueAtTime(950, t + 0.04);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.22, t + 0.003);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.045);
      osc.connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.05);
    } else {
      // Мягкий пузырек
      osc.type = 'sine';
      osc.frequency.setValueAtTime(260, t);
      osc.frequency.exponentialRampToValueAtTime(620, t + 0.03);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.2, t + 0.002);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.035);
      osc.connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.04);
    }
  }

  // Деликатный перестук скролла
  playScroll() {
    this.initAudio();
    const t = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = 'triangle';
    osc.frequency.setValueAtTime(980, t);
    osc.frequency.exponentialRampToValueAtTime(160, t + 0.016);
    gain.gain.setValueAtTime(0.0001, t);
    gain.gain.linearRampToValueAtTime(0.04, t + 0.001);
    gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.018);
    osc.connect(gain).connect(this.master);
    osc.start(t); osc.stop(t + 0.02);
  }

  bindEvents() {
    if (this.opt.clicks) {
      document.addEventListener('pointerdown', (e) => {
        if (e.target.closest('button, a, input, select, [role="button"], .card, i')) {
          this.playClick('wood');
        }
      }, { passive: true });
    }
    if (this.opt.scroll) {
      let lastY = window.scrollY, lastT = Date.now();
      window.addEventListener('scroll', () => {
        const y = window.scrollY;
        const diff = Math.abs(y - lastY);
        const now = Date.now();
        if (diff > 45 && (now - lastT) > 40) {
          this.playScroll();
          lastY = y; lastT = now;
        }
      }, { passive: true });
    }
  }
}

// Запуск по умолчанию
ElektromirASMR.start({ season: 'autumn', volume: 0.45, clicks: true, scroll: true });
</script>
