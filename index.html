Вставьте данный скрипт перед тегом </body>. Он не требует внешних библиотек и аудио-файлов. 

<!-- ASMR Pure Elements Engine (Neon Pastel Fuchsia Edition) -->
<script>
class ElektromirASMR {
  static start(options = {}) {
    if (window._elektromirASMR) return window._elektromirASMR;
    window._elektromirASMR = new ElektromirASMR(options);
    return window._elektromirASMR;
  }

  constructor(opts) {
    this.opt = Object.assign({
      wind: true, rain: true, stream: true,
      windVol: 0.5, rainVol: 0.6, streamVol: 0.45,
      masterVol: 0.5, clicks: true, scroll: true, haptic: true
    }, opts);
    this.ctx = null;
    this.master = null;
    this.bindEvents();
  }

  initAudio() {
    if (!this.ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      this.ctx = new AC();
      this.master = this.ctx.createGain();
      this.master.gain.setValueAtTime(this.opt.masterVol, this.ctx.currentTime);
      this.master.connect(this.ctx.destination);
    }
    if (this.ctx.state === 'suspended') this.ctx.resume();
  }

  vibrate(pattern = 15) {
    if (this.opt.haptic && 'vibrate' in navigator) {
      try { navigator.vibrate(pattern); } catch (e) {}
    }
  }

  playClick(type = 'wood') {
    this.initAudio();
    this.vibrate(14);
    const t = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    if (type === 'wood') {
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(540, t);
      osc.frequency.exponentialRampToValueAtTime(115, t + 0.035);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.28, t + 0.002);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.04);
      osc.connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.045);
    } else if (type === 'drop') {
      osc.type = 'sine';
      osc.frequency.setValueAtTime(380, t);
      osc.frequency.exponentialRampToValueAtTime(1050, t + 0.042);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.24, t + 0.003);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.048);
      osc.connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.05);
    } else {
      osc.type = 'sine';
      osc.frequency.setValueAtTime(260, t);
      osc.frequency.exponentialRampToValueAtTime(680, t + 0.028);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.22, t + 0.002);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.035);
      osc.connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.04);
    }
  }

  playScroll() {
    this.initAudio();
    this.vibrate(8);
    const t = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = 'triangle';
    osc.frequency.setValueAtTime(960, t);
    osc.frequency.exponentialRampToValueAtTime(150, t + 0.016);
    gain.gain.setValueAtTime(0.0001, t);
    gain.gain.linearRampToValueAtTime(0.04, t + 0.001);
    gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.018);
    osc.connect(gain).connect(this.master);
    osc.start(t); osc.stop(t + 0.02);
  }

  bindEvents() {
    if (this.opt.clicks) {
      document.addEventListener('pointerdown', (e) => {
        if (e.target.closest('button, a, input, select, [role="button"], .card, i')) {
          this.playClick('wood');
        }
      }, { passive: true });
    }
    if (this.opt.scroll) {
      let lastY = window.scrollY, lastT = Date.now();
      window.addEventListener('scroll', () => {
        const y = window.scrollY;
        const diff = Math.abs(y - lastY);
        const now = Date.now();
        if (diff > 45 && (now - lastT) > 50) {
          this.playScroll();
          lastY = y; lastT = now;
        }
      }, { passive: true });
    }
  }
}

// Запуск по умолчанию
ElektromirASMR.start();
</script>
