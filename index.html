<!-- Добавьте перед закрывающим тегом </body> в index.html -->
<script>
/**
 * Ultra-realistic Procedural Seasonal ASMR Engine
 * GitHub Pages: https://romanmudrenov.github.io/elektromir/
 */
class SeasonalASMR {
  static init(config = { season: 'autumn', scroll: true, tap: true, volume: 0.5 }) {
    if (window._seasonalAsmrInstance) return window._seasonalAsmrInstance;
    const instance = new SeasonalASMR(config);
    window._seasonalAsmrInstance = instance;
    return instance;
  }

  constructor(cfg) {
    this.cfg = Object.assign({ season: 'autumn', scroll: true, tap: true, volume: 0.5 }, cfg);
    this.ctx = null;
    this.master = null;
    this.setupTriggers();
  }

  ensureContext() {
    if (!this.ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      this.ctx = new AC();
      this.master = this.ctx.createGain();
      this.master.gain.setValueAtTime(this.cfg.volume, this.ctx.currentTime);
      this.master.connect(this.ctx.destination);
    }
    if (this.ctx.state === 'suspended') this.ctx.resume();
  }

  playTap(variant = 'primary') {
    this.ensureContext();
    const t = this.ctx.currentTime;
    const s = this.cfg.season;
    
    // Процедурный синтез под сезон
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    const filter = this.ctx.createBiquadFilter();

    if (s === 'spring') {
      // Весенняя роса / капля (синус с восходящим глиссандо)
      osc.type = 'sine';
      osc.frequency.setValueAtTime(380, t);
      osc.frequency.exponentialRampToValueAtTime(1100, t + 0.04);
      filter.type = 'bandpass';
      filter.frequency.setValueAtTime(800, t);
      filter.Q.setValueAtTime(4, t);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.3, t + 0.004);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.045);
      osc.connect(filter).connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.05);
    } else if (s === 'summer') {
      // Летняя веточка (короткий щелчок с резонансом дерева)
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(540, t);
      osc.frequency.exponentialRampToValueAtTime(120, t + 0.035);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.28, t + 0.002);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.038);
      osc.connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.04);
    } else if (s === 'autumn') {
      // Осенний шорох / капля дождя
      osc.type = 'sine';
      osc.frequency.setValueAtTime(900, t);
      osc.frequency.exponentialRampToValueAtTime(280, t + 0.03);
      filter.type = 'bandpass'; filter.frequency.setValueAtTime(1600, t); filter.Q.setValueAtTime(2.5, t);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.25, t + 0.002);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.032);
      osc.connect(filter).connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.035);
    } else {
      // Зимний хруст наста / кристалл
      osc.type = 'sine';
      osc.frequency.setValueAtTime(1800, t);
      osc.frequency.exponentialRampToValueAtTime(950, t + 0.04);
      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.2, t + 0.001);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.045);
      osc.connect(gain).connect(this.master);
      osc.start(t); osc.stop(t + 0.05);
    }
  }

  playScrollTick() {
    this.ensureContext();
    const t = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = 'triangle';
    osc.frequency.setValueAtTime(1200, t);
    osc.frequency.exponentialRampToValueAtTime(140, t + 0.015);
    gain.gain.setValueAtTime(0.0001, t);
    gain.gain.linearRampToValueAtTime(0.05, t + 0.002);
    gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.016);
    osc.connect(gain).connect(this.master);
    osc.start(t); osc.stop(t + 0.018);
  }

  setupTriggers() {
    if (this.cfg.tap) {
      document.addEventListener('pointerdown', (e) => {
        if (e.target.closest('button, a, input, select, [role="button"], i, .card, .btn')) {
          this.playTap();
        }
      }, { passive: true });
    }
    if (this.cfg.scroll) {
      let lastY = window.scrollY, lastT = Date.now();
      window.addEventListener('scroll', () => {
        const y = window.scrollY;
        const d = Math.abs(y - lastY);
        const now = Date.now();
        if (d > 40 && (now - lastT) > 45) {
          this.playScrollTick();
          lastY = y; lastT = now;
        }
      }, { passive: true });
    }
  }
}

// Запуск: выберите season ('spring' | 'summer' | 'autumn' | 'winter')
SeasonalASMR.init({ season: 'autumn', scroll: true, tap: true, volume: 0.4 });
</script>
